// DAX: Budget vs Actual Revenue Variance & Year-over-Year Growth
// Target Variance Calculation
Sales Variance % = 
VAR ActualSales = [Total Sales]
VAR BudgetedSales = [Total Budget]
RETURN
DIVIDE(
    ActualSales - BudgetedSales,
    BudgetedSales,
    0
)

// YTD Time Intelligence
Total Sales YTD = 
CALCULATE(
    [Total Sales],
    DATESYTD('Calendar'[Date])
)

    TypedColumns = Table.TransformColumnTypes(CleanedNulls, {
        {"OrderDate", type date}, 
        {"SalesAmount", Currency.Type}, 
        {"CustomerID", type text}
    })

    TypedColumns
